699 static std::tuple<Scalars, Scalars, RealVectors, RealVectors>
700 getCNCPrincipalCurvaturesAndDirections
701 ( CountedPtr<typename Base::SurfaceMesh> mesh,
702 const Parameters& params = parametersShapeGeometry() )
703 {
704 bool unit_u = params["unit_u"].as<int>();
705 CNCComputer computer(*mesh, unit_u);
706
707 const auto& mu0 = computer.computeMu0().face_measures;
708 const auto& muxy = computer.computeMuXY().face_measures;
709
710 if (mu0.size() != muxy.size())
711 {
712 trace.error() << "Error mu0 and muxy does not have same size. \n";
713 return {};
714 }
715
716 if (mesh->faceNormals().size() == 0)
717 {
718 // Try to use vertex normals if any
719 if (mesh->vertexNormals().size() == 0)
720 mesh->computeFaceNormalsFromPositions();
721 else
722 mesh->computeFaceNormalsFromVertexNormals();
723 }
724
725 if (mesh->faceNormals().size() != mu0.size())
726 {
727 trace.error() << "Error face normals does not have same size as mu0.\n";
728 return {};
729 }
730
731 Scalars k1(mu0.size()), k2(mu0.size());
732 RealVectors d1(mu0.size()), d2(mu0.size());
733
734 const auto& normals = mesh->faceNormals();
735 for (size_t i = 0; i < mu0.size(); ++i)
736 std::tie(k1[i], k2[i], d1[i], d2[i]) = CNCComputer::principalCurvatures(mu0[i], muxy[i], normals[i]);
737
738 return std::make_tuple(k1, k2, d1, d2);
739 }
